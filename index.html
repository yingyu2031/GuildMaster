function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("成員名單") || ss.getSheets()[0]; var historySheet = ss.getSheetByName("異動歷程"); var data = {}; if (e && e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch(err) { data = e.parameter; } } else if (e && e.parameter) { data = e.parameter; } var lineUid = data.lineUserId || ""; var gameNick = data.gameNickname || ""; var gameClass = data.gameClass || ""; var lineName = data.lineDisplayName || ""; var allowSearch = (data.allowSearch === "開啟" || data.allowSearch === true || data.allowSearch === "Y") ? "Y" : "N"; var nowTime = Utilities.formatDate(new Date(), "Asia/Taipei", "yyyy-MM-dd HH:mm:ss"); if (!lineUid) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "缺少 LINE UID" })) .setMimeType(ContentService.MimeType.JSON); } var rows = sheet.getDataRange().getValues(); var rowIndex = -1; // A 欄為 LINE UID for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === String(lineUid).trim()) { rowIndex = i + 1; break; } } if (rowIndex > 0) { // 取出既有舊值 var oldNick = rows[rowIndex - 1][1]; var oldClass = rows[rowIndex - 1][2]; var oldLineName = rows[rowIndex - 1][3]; var oldSearch = rows[rowIndex - 1][4]; // 若有異動歷程分頁，逐欄比對並寫入歷程 (A: 時間, B: UID, C: 欄位, D: 原值, E: 新值) if (historySheet) { if (String(oldNick) !== String(gameNick)) { historySheet.appendRow([nowTime, lineUid, "遊戲暱稱", oldNick, gameNick]); } if (String(oldClass) !== String(gameClass)) { historySheet.appendRow([nowTime, lineUid, "遊戲職業", oldClass, gameClass]); } if (String(oldLineName) !== String(lineName)) { historySheet.appendRow([nowTime, lineUid, "LINE 暱稱", oldLineName, lineName]); } if (String(oldSearch) !== String(allowSearch)) { historySheet.appendRow([nowTime, lineUid, "提供查詢", oldSearch, allowSearch]); } } // 更新成員名單 sheet.getRange(rowIndex, 2).setValue(gameNick); sheet.getRange(rowIndex, 3).setValue(gameClass); sheet.getRange(rowIndex, 4).setValue(lineName); sheet.getRange(rowIndex, 5).setValue(allowSearch); sheet.getRange(rowIndex, 7).setValue(nowTime); } else { // 新會員註冊 sheet.appendRow([lineUid, gameNick, gameClass, lineName, allowSearch, "待審核", nowTime]); // 新增註冊歷程紀錄 if (historySheet) { historySheet.appendRow([nowTime, lineUid, "新會員註冊", "-", gameNick + " (" + gameClass + ")"]); } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doGet(e) { try { var lineUid = e.parameter.uid; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("成員名單") || ss.getSheets()[0]; var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === String(lineUid).trim()) { var result = { status: "found", lineUserId: rows[i][0], gameNickname: rows[i][1], gameClass: rows[i][2], lineDisplayName: rows[i][3], allowSearch: (rows[i][4] === "Y" || rows[i][4] === "開啟" || rows[i][4] === true), accountStatus: rows[i][5] }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ status: "not_found" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
