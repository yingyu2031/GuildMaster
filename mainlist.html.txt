        ];

        let currentDate = new Date().toISOString().split('T')[0];

        window.onload = function() {
            document.getElementById('rosterDate').value = currentDate;
            document.getElementById('rosterDate').addEventListener('change', (e) => {
                currentDate = e.target.value;
                loadRosterForDate(currentDate);
            });

            document.getElementById('searchInput').addEventListener('input', renderRosterPool);
            document.getElementById('classFilter').addEventListener('change', renderRosterPool);

            // Populate class filter options
            const classes = Object.keys(classColors);
            const select = document.getElementById('classFilter');
            classes.forEach(c => {
                const opt = document.createElement('option');
                opt.value = c;
                opt.textContent = c;
                select.appendChild(opt);
            });

            renderAll();
        };

        function renderAll() {
            renderRosterPool();
            renderSquads();
            renderMergedTeams();
            renderLeaveList();
            updateStats();
        }

        function renderRosterPool() {
            const container = document.getElementById('rosterPoolContainer');
            container.innerHTML = '';

            const keyword = document.getElementById('searchInput').value.trim().toLowerCase();
            const selectedClass = document.getElementById('classFilter').value;

            // Get all assigned member names
            const assignedMembers = new Set();
            squads.forEach(s => s.members.forEach(m => assignedMembers.add(m)));

            let filtered = members.filter(m => {
                const matchKeyword = m.name.toLowerCase().includes(keyword);
                const matchClass = selectedClass === 'all' || m.class === selectedClass;
                return matchKeyword && matchClass;
            });

            document.getElementById('memberCountInfo.textContent` = `總計 ${filtered.length} 人`;

            // Group by class or display list
            filtered.forEach(m => {
                const isAssigned = assignedMembers.has(m.name);
                const div = document.createElement('div');
                div.className = `flex items-center justify-between p-2 rounded-xl border border-bordercolor bg-slate-900/60 transition hover:border-slate-500 ${isAssigned ? 'opacity-50' : 'cursor-grab'}`;
                div.draggable = !isAssigned;
                
                if (!isAssigned) {
                    div.addEventListener('dragstart', (e) => {
                        e.dataTransfer.setData('text/plain', m.name);
                        div.classList.add('dragging');
                    });
                    div.addEventListener('dragend', () => div.classList.remove('dragging'));
                }

                const badgeColor = classColors[m.class] || 'bg-slate-700 text-slate-300';

                div.innerHTML = `
                    <div class="flex items-center gap-2">
                        <span class="text-xs px-2 py-0.5 rounded border ${badgeColor} font-medium">${m.class}</span>
                        <span class="text-sm font-medium text-slate-200">${m.name}</span>
                    </div>
                    <div class="flex items-center gap-1">
                        ${isAssigned ? '<span class="text-[10px] bg-slate-800 text-slate-400 px-1.5 py-0.5 rounded">已排入</span>' : '<i class="fa-solid fa-grip-vertical text-slate-600 text-xs"></i>'}
                    </div>
                `;
                container.appendChild(div);
            });

            // Also render unassigned bench
            renderUnassignedPool(assignedMembers);
        }

        function renderUnassignedPool(assignedMembers) {
            const container = document.getElementById('unassignedContainer');
            container.innerHTML = '';
            
            const unassigned = members.filter(m => !assignedMembers.has(m.name));
            document.getElementById('unassignedCount').textContent = unassigned.length;

            if (unassigned.length === 0) {
                container.innerHTML = '<span class="text-xs text-slate-500 italic p-1">無未分配人員</span>';
                return;
            }

            unassigned.forEach(m => {
                const tag = document.createElement('div');
                const badgeColor = classColors[m.class] || 'bg-slate-700 text-slate-300';
                tag.className = `text-xs px-2.5 py-1 rounded-lg border ${badgeColor} cursor-grab flex items-center gap-1.5 bg-slate-900 shadow-sm`;
                tag.draggable = true;
                tag.addEventListener('dragstart', (e) => {
                    e.dataTransfer.setData('text/plain', m.name);
                });
                tag.innerHTML = `<span>${m.name}</span><span class="text-[10px] opacity-70">(${m.class})</span>`;
                container.appendChild(tag);
            });
        }

        function renderSquads() {
            const grid = document.getElementById('squadsGrid');
            grid.innerHTML = '';

            squads.forEach((squad, index) => {
                const card = document.createElement('div');
                card.className = 'bg-cardbg border border-bordercolor rounded-2xl p-4 shadow-lg flex flex-col transition hover:border-slate-600';
                card.addEventListener('dragover', (e) => {
                    e.preventDefault();
                    card.classList.add('drag-over');
                });
                card.addEventListener('dragleave', () => {
                    card.classList.remove('drag-over');
                });
                card.addEventListener('drop', (e) => {
                    e.preventDefault();
                    card.classList.remove('drag-over');
                    const memberName = e.dataTransfer.getData('text/plain');
                    if (memberName) {
                        assignMemberToSquad(memberName, squad.id);
                    }
                });

                // Header with editable name
                card.innerHTML = `
                    <div class="flex items-center justify-between mb-3 pb-2 border-b border-bordercolor gap-2">
                        <div class="flex items-center gap-2 flex-1">
                            <span class="text-xs bg-blue-600/30 text-blue-400 font-bold px-2 py-0.5 rounded border border-blue-500/30">#${index + 1}</span>
                            <input type="text" value="${squad.name}" onblur="updateSquadName('${squad.id}', this.value)" class="bg-transparent text-sm font-bold text-slate-200 focus:outline-none focus:bg-slate-900 px-1.5 py-0.5 rounded w-full border border-transparent focus:border-bordercolor transition">
                        </div>
                        <div class="flex items-center gap-1">
                            <span class="text-xs text-slate-400 font-medium px-2 py-0.5 bg-slate-900 rounded">${squad.members.length} 人</span>
                            <button onclick="deleteSquad('${squad.id}')" class="text-slate-500 hover:text-rose-400 p-1 text-xs transition" title="刪除隊伍">
                                <i class="fa-solid fa-trash-can"></i>
                            </button>
                        </div>
                    </div>
                    <div class="flex-1 min-h-[110px] bg-slate-950/60 rounded-xl p-2.5 border border-bordercolor flex flex-wrap gap-2 content-start overflow-y-auto max-h-[180px]" id="squad-members-${squad.id}">
                        <!-- Member tags inside squad -->
                    </div>
                `;

                const memberContainer = card.querySelector(`#squad-members-${squad.id}`);
                if (squad.members.length === 0) {
                    memberContainer.innerHTML = '<div class="w-full h-full flex items-center justify-center text-xs text-slate-600 italic py-6">請將左側人員拖曳至此處</div>';
                } else {
                    squad.members.forEach(mName => {
                        const memberObj = members.find(m => m.name === mName) || { name: mName, class: "未知" };
                        const badgeColor = classColors[memberObj.class] || 'bg-slate-700 text-slate-300';
                        
                        const tag = document.createElement('div');
                        tag.className = `text-xs px-2.5 py-1 rounded-xl border ${badgeColor} flex items-center gap-2 bg-slate-900 shadow-sm animate-fadeIn`;
                        tag.innerHTML = `
                            <span class="font-semibold">${memberObj.name}</span>
                            <span class="text-[10px] opacity-70">(${memberObj.class})</span>
                            <button onclick="removeMemberFromSquad('${squad.id}', '${memberObj.name}')" class="hover:text-rose-400 transition ml-0.5">
                                <i class="fa-solid fa-xmark"></i>
                            </button>
                        `;
                        memberContainer.appendChild(tag);
                    });
                }

                grid.appendChild(card);
            });
        }

        function assignMemberToSquad(memberName, squadId) {
            // Remove from other squads first
            squads.forEach(s => {
                s.members = s.members.filter(m => m !== memberName);
            });
            // Add to target squad
            const target = squads.find(s => s.id === squadId);
            if (target) {
                target.members.push(memberName);
            }
            renderAll();
        }

        function removeMemberFromSquad(squadId, memberName) {
            const squad = squads.find(s => s.id === squadId);
            if (squad) {
                squad.members = squad.members.filter(m => m !== memberName);
            }
            renderAll();
        }

        function dropToUnassigned(e) {
            e.preventDefault();
            const memberName = e.dataTransfer.getData('text/plain');
            if (memberName) {
                squads.forEach(s => {
                    s.members = s.members.filter(m => m !== memberName);
                });
                renderAll();
            }
        }

        function allowDrop(e) {
            e.preventDefault();
        }

        function removeDropStyle(e) {
            // helper
        }

        function updateSquadName(squadId, newName) {
            const squad = squads.find(s => s.id === squadId);
            if (squad && newName.trim() !== "") {
                squad.name = newName.trim();
            }
        }

        function addNewSquad() {
            const newId = 's' + (Date.now());
            const newNum = squads.length + 1;
            squads.push({
                id: newId,
                name: `陣伍${newNum} (新)`,
                members: []
            });
            renderAll();
        }

        function deleteSquad(squadId) {
            if (squads.length <= 1) {
                showMsg("提示", "至少需保留一個隊伍！", "warning");
                return;
            }
            squads = squads.filter(s => s.id !== squadId);
            // Also clean up merged teams
            mergedTeams.forEach(mt => {
                mt.squadIds = mt.squadIds.filter(id => id !== squadId);
            });
            mergedTeams = mergedTeams.filter(mt => mt.squadIds.length > 0);
            renderAll();
        }

        function resetAllAssignments() {
            squads.forEach(s => s.members = []);
            mergedTeams = [];
            renderAll();
            showMsg("成功", "已重置所有隊伍分配！", "success");
        }

        function openMergeModal() {
            const container = document.getElementById('mergeSquadCheckboxes');
            container.innerHTML = '';
            squads.forEach(s => {
                const div = document.createElement('label');
                div.className = 'flex items-center gap-3 text-sm text-slate-200 cursor-pointer hover:bg-slate-900 p-1.5 rounded-lg transition';
                div.innerHTML = `
                    <input type="checkbox" value="${squad.id}" class="rounded border-bordercolor bg-slate-900 text-purple-600 focus:ring-purple-500 w-4 h-4">
                    <span class="font-medium">${squad.name}</span>
                    <span class="text-xs text-slate-400 ml-auto">(${squad.members.length} 人)</span>
                `;
                container.appendChild(div);
            });
            document.getElementById('mergeTeamName').value = '';
            document.getElementById('mergeModal').classList.remove('hidden');
        }

        function closeMergeModal() {
            document.getElementById('mergeModal').classList.add('hidden');
        }

        function confirmMergeTeams() {
            const teamName = document.getElementById('mergeTeamName').value.trim();
            const checkboxes = document.querySelectorAll('#mergeSquadCheckboxes input[type="checkbox"]:checked');
            const squadIds = Array.from(checkboxes).map(cb => cb.value);

            if (!teamName) {
                showMsg("提示", "請輸入合併後的大團隊名稱！", "warning");
                return;
            }
            if (squadIds.length < 2) {
                showMsg("提示", "請至少勾選兩個隊伍進行合併！", "warning");
                return;
            }

            mergedTeams.push({
                name: teamName,
                squadIds: squadIds
            });

            closeMergeModal();
            renderMergedTeams();
            showMsg("成功", `成功建立合併團隊：${teamName}`, "success");
        }

        function renderMergedTeams() {
            const container = document.getElementById('mergedTeamsContainer');
            container.innerHTML = '';

            if (mergedTeams.length === 0) return;

            mergedTeams.forEach((mt, idx) => {
                // Gather all members from these squads
                let allMembers = [];
                mt.squadIds.forEach(sid => {
                    const sq = squads.find(s => s.id === sid);
                    if (sq) allMembers = allMembers.concat(sq.members);
                });

                const div = document.createElement('div');
                div.className = 'bg-cardbg border border-purple-500/40 rounded-2xl p-4 shadow-lg space-y-3';
                div.innerHTML = `
                    <div class="flex items-center justify-between">
                        <div class="flex items-center gap-2">
                            <div class="bg-purple-600/30 text-purple-300 px-2.5 py-1 rounded-xl border border-purple-500/40 font-bold text-sm flex items-center gap-1.5">
                                <i class="fa-solid fa-object-ungroup"></i> ${mt.name}
                            </div>
                            <span class="text-xs text-slate-400">包含隊伍: ${mt.squadIds.map(sid => {
                                const s = squads.find(sq => sq.id === sid);
                                return s ? s.name : '';
                            }).filter(Boolean).join(' + ')}</span>
                        </div>
                        <div class="flex items-center gap-2">
                            <span class="text-xs bg-purple-500/20 text-purple-300 px-2.5 py-1 rounded-full border border-purple-500/30 font-medium">總成員: ${allMembers.length} 人</span>
                            <button onclick="removeMergedTeam(${idx})" class="text-slate-500 hover:text-rose-400 p-1 text-xs transition"><i class="fa-solid fa-trash-can"></i></button>
                        </div>
                    </div>
                    <div class="flex flex-wrap gap-1.5 bg-slate-950/50 p-2.5 rounded-xl border border-bordercolor">
                        ${allMembers.length === 0 ? '<span class="text-xs text-slate-500 italic">目前合併團隊無成員</span>' : allMembers.map(mName => {
                            const mObj = members.find(m => m.name === mName) || { name: mName, class: "未知" };
                            const badgeColor = classColors[mObj.class] || 'bg-slate-700 text-slate-300';
                            return `<span class="text-xs px-2.5 py-1 rounded-lg border ${badgeColor} bg-slate-900">${mObj.name} <span class="opacity-70 text-[10px]">(${mObj.class})</span></span>`;
                        }).join('')}
                    </div>
                `;
                container.appendChild(div);
            });
        }

        function removeMergedTeam(idx) {
            mergedTeams.splice(idx, 1);
            renderMergedTeams();
        }

        function renderLeaveList() {
            const container = document.getElementById('leaveListContainer');
            container.innerHTML = '';
            
            // Filter leaves matching current date or show all
            const countBadge = document.getElementById('leaveCountBadge');
            countBadge.textContent = leaveList.length;

            if (leaveList.length === 0) {
                container.innerHTML = '<div class="text-center py-6 text-xs text-slate-500 italic">目前無請假紀錄</div>';
                return;
            }

            leaveList.forEach((l, idx) => {
                const div = document.createElement('div');
                div.className = 'flex items-center justify-between p-3 bg-slate-950/60 border border-bordercolor rounded-xl';
                div.innerHTML = `
                    <div class="flex items-center gap-3">
                        <span class="text-xs bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded border border-amber-500/30 font-medium">${l.class}</span>
                        <span class="text-sm font-semibold text-slate-200">${l.name}</span>
                        <span class="text-xs text-slate-400"><i class="fa-regular fa-calendar mr-1"></i>${l.date}</span>
                    </div>
                    <button onclick="deleteLeavePerson(${idx})" class="text-slate-500 hover:text-rose-400 text-xs p-1 transition"><i class="fa-solid fa-trash-can"></i></button>
                `;
                container.appendChild(div);
            });
        }

        function addLeavePerson() {
            const name = document.getElementById('leaveNameInput').value.trim();
            const cls = document.getElementById('leaveClassInput').value.trim();
            const date = document.getElementById('leaveDateInput').value || currentDate;

            if (!name || !cls) {
                showMsg("提示", "請填寫請假人員名稱與職業！", "warning");
                return;
            }

            leaveList.push({ name, class: cls, date });
            document.getElementById('leaveNameInput').value = '';
            document.getElementById('leaveClassInput').value = '';
            renderLeaveList();
            showMsg("成功", "已新增請假紀錄！", "success");
        }

        function deleteLeavePerson(idx) {
            leaveList.splice(idx, 1);
            renderLeaveList();
        }

        function openLeaveModal() {

            document.getElementById('leaveModal').classList.remove('hidden');
        }

        // Save & History with localStorage
        function saveRoster() {
            const dataToSave = {
                date: currentDate,
                squads: squads,
                mergedTeams: mergedTeams,
                leaveList: leaveList,
                updatedAt: new Date().toLocaleString()
            };

            localStorage.setItem(`guild_roster_${currentDate}`, JSON.stringify(dataToSave));
            showMsg("成功", `日期 [${currentDate}] 的主戰名單已成功存檔！`, "success");
        }

        function openHistoryModal() {
            const container = document.getElementById('historyListContainer');
            container.innerHTML = '';

            // Find all saved roster keys in localStorage
            const keys = [];
            for (let i = 0; i < localStorage.length; i++) {
                const key = localStorage.key(i);
                if (key.startsWith('guild_roster_')) {
                    keys.push(key);
                }
            }

            if (keys.length === 0) {
                container.innerHTML = '<div class="text-center py-8 text-xs text-slate-500 italic">尚無任何歷史存檔紀錄</div>';
            } else {
                keys.sort().reverse().forEach(key => {
                    const data = JSON.parse(localStorage.getItem(key));
                    const div = document.createElement('div');
                    div.className = 'flex items-center justify-between p-3 bg-slate-900/80 border border-bordercolor rounded-xl hover:border-blue-500/50 transition cursor-pointer';
                    div.onclick = () => loadHistoryData(data);
                    div.innerHTML = `
                        <div>
                            <div class="text-sm font-bold text-slate-200 flex items-center gap-2">
                                <i class="fa-regular fa-calendar-days text-blue-400"></i> ${data.date}
                            </div>
                            <div class="text-xs text-slate-400 mt-0.5">存檔時間: ${data.updatedAt || '未知'} | 隊伍數: ${data.squads.length}</div>
                        </div>
                        <span class="text-xs bg-blue-600/20 text-blue-400 border border-blue-500/30 px-3 py-1.5 rounded-lg font-medium">載入此排班</span>
                    `;
                    container.appendChild(div);
                });
            }

            document.getElementById('historyModal').classList.remove('hidden');
        }

        function closeHistoryModal() {
            document.getElementById('historyModal').classList.add('hidden');
        }

        function loadHistoryData(data) {
            currentDate = data.date;
            document.getElementById('rosterDate').value = currentDate;
            squads = data.squads || squads;
            mergedTeams = data.mergedTeams || [];
            leaveList = data.leaveList || leaveList;
            closeHistoryModal();
            renderAll();
            showMsg("成功", `已成功載入 [${currentDate}] 的排班名單！`, "success");
        }

        function loadRosterForDate(dateStr) {
            const saved = localStorage.getItem(`guild_roster_${dateStr}`);
            if (saved) {
                const data = JSON.parse(saved);
                squads = data.squads || squads;
                mergedTeams = data.mergedTeams || [];
                leaveList = data.leaveList || leaveList;
                renderAll();
                showMsg("提示", `已自動載入日期 [${dateStr}] 的存檔排班。`, "info");
            } else {
                // If not saved, keep current layout or reset
                renderAll();
            }
        }

        function updateStats() {
            let totalAssigned = 0;
            squads.forEach(s => totalAssigned += s.members.length);
            document.getElementById('totalAssignedBadge').textContent = `已配置: ${totalAssigned} 人`;
        }

        // Custom Message Box
        function showMsg(title, content, type = 'info') {
            document.getElementById('msgTitle').textContent = title;
            document.getElementById('msgContent').textContent = content;
            const iconEl = document.getElementById('msgIcon');
            if (type === 'success') {
                iconEl.innerHTML = '<i class="fa-solid fa-circle-check text-emerald-400"></i>';
            } else if (type === 'warning') {
                iconEl.innerHTML = '<i class="fa-solid fa-triangle-exclamation text-amber-400"></i>';
            } else {
                iconEl.innerHTML = '<i class="fa-solid fa-circle-info text-blue-400"></i>';
            }
            document.getElementById('msgModal').classList.remove('hidden');
        }

        function closeMsgModal() {
            document.getElementById('msgModal').classList.add('hidden');
        }
